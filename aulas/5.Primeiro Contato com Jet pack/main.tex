\section{O que é Jetpack Compose?}

\begin{frame}{O que é Jetpack Compose?}
  \begin{itemize}
    \item É o framework moderno do Android para criação de interfaces de usuário.
    \item Usa a linguagem Kotlin de forma nativa.
    \item Substitui o sistema tradicional de Views (XML).
  \end{itemize}

  \begin{block}{Em resumo}
    Compose transforma o modo como criamos interfaces Android.
    \begin{itemize}
      \item Menos código.
      \item Mais rápido (visualmente).
      \item Código mais legível.
    \end{itemize}
  \end{block}

  \begin{alertblock}{Não é uma biblioteca, é um sistema de UI Declarativo}
    \begin{itemize}
      \item Em vez de dizer 
      \textbf{como} desenhar cada componente (imperativo),
      \pause
      \item Você apenas \textbf{descreve} como a tela deve ser para cada estado
      \pause
      \item O Compose cuida de desenhar e atualizar a tela automaticamente.
    \end{itemize}
  \end{alertblock}
\end{frame}

\begin{frame}{UI Declarativa vs UI Imperativa}
  \begin{exampleblock}{Pensamento moderno}
    \begin{minipage}{0.48\textwidth}
      \textbf{UI Imperativa (ViewSystem)}
      \begin{itemize}
        \item Você atualiza elementos da tela manualmente.
        \item Exemplo: \texttt{TextView.setText("Olá")}
        \item Se a tela tiver 3 estados, você escreve código para cada estado.
      \end{itemize}
    \end{minipage}
    \hfill
    \begin{minipage}{0.48\textwidth}
      \textbf{UI Declarativa (Compose)}
      \begin{itemize}
        \item Você descreve a tela para cada estado.
        \item Exemplo: Função \texttt{MeuTexto(texto="Olá")}
        \item Se a tela tiver 3 estados, você escreve 3 funções (ou uma lógica).
      \end{itemize}
    \end{minipage}
  \end{exampleblock}
\end{frame}

\begin{frame}[fragile]{Carregamento de Imagens e Modelo de Dados}
  A função auxiliar \texttt{carregarImagemAsset} e a data class \texttt{Docente}.
  \begin{lstlisting}[style=KotlinStyle]
data class Docente(
    val nome: String, val titulacao: String,
    val lattes: String, val email: String, val foto: String
)

@Composable
fun carregarImagemAsset(nomeArquivo: String): ImageBitmap? {
    val context = LocalContext.current
    return try {
        context.assets.open(nomeArquivo).use { inputStream ->
            BitmapFactory.decodeStream(inputStream)?.asImageBitmap()
        }
    } catch (e: Exception) { null }
}
  \end{lstlisting}
\end{frame}

\begin{frame}[fragile]{O Componente PesquisadorCard}
  Exibe os dados de um docente individualmente.
  \begin{lstlisting}[style=KotlinStyle, basicstyle=\ttfamily\tiny]
@Composable
fun PesquisadorCard(nome: String, titulacao: String, lattes: String, email: String, foto: String, modifier: Modifier = Modifier) {
    val imagem = carregarImagemAsset(foto)
    Card(modifier = modifier.fillMaxWidth().padding(16.dp), colors = CardDefaults.cardColors(containerColor = Color(0xFFF1F4F5))) {
        Column(modifier = Modifier.width(1200.dp).height(300.dp), horizontalAlignment = Alignment.CenterHorizontally) {
            imagem?.let { Image(bitmap = it, contentDescription = "Foto", modifier = Modifier.fillMaxWidth().height(120.dp), contentScale = ContentScale.Crop) }
            Text(text = nome, fontSize = 12.sp, fontWeight = FontWeight.Bold, color = Color(0xFF222222))
            Text(text = titulacao, fontSize = 12.sp, color = Color(0xFF666666))
            Text(text = lattes, fontSize = 12.sp, color = Color(0xFF007AFF))
            Text(text = email, fontSize = 12.sp, color = Color(0xFF444444))
        }
    }
}
  \end{lstlisting}
\end{frame}

\begin{frame}[fragile]{Listando os Docentes com Grid}
  \texttt{LazyVerticalGrid} organiza os cartões em colunas.
  \begin{lstlisting}[style=KotlinStyle, basicstyle=\ttfamily\tiny]
@Composable
fun ListaPesquisadores(){
    val docentes = listOf(/* instâncias de Docente omitidas por espaço */)
    LazyVerticalGrid(
        columns = GridCells.Fixed(2),
        modifier = Modifier.fillMaxSize(),
        contentPadding = PaddingValues(16.dp),
        horizontalArrangement = Arrangement.spacedBy(16.dp),
        verticalArrangement = Arrangement.spacedBy(16.dp)
    ) {
        items(docentes) { docente ->
            PesquisadorCard(
                nome = docente.nome,
                titulacao = docente.titulacao,
                lattes = docente.lattes,
                email = docente.email,
                foto = docente.foto
            )
        }
    }
}
  \end{lstlisting}
\end{frame}
